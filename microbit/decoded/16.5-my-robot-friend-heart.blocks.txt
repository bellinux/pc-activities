para siempre:
  mostrar LEDs:
      . # . # .
      # # # # #
      # # # # #
      . # # # .
      . . # . .
  reproduce secuencia tono Re5 durante 1/8 pulso en modo en segundo plano
  pausa (ms) pausaLarga   // Se utiliza la variable "pausaLarga" para cambiar la velocidad de latido del corazón grande dependiendo de la posición del dispositivo.
  reproduce secuencia tono Re5 durante 1/8 pulso en modo en segundo plano
  repetir 4 veces:
    mostrar LEDs:
        . # . # .
        # . # . #
        # . . . #
        . # . # .
        . . # . .
    pausa (ms) pausaCorta   // Se utiliza la variable "pausaCorta" para cambiar la velocidad de latido del corazón chico dependiendo de la posición del dispositivo.
    mostrar LEDs:
        . . . . .
        . # . # .
        . # # # .
        . . # . .
        . . . . .
    pausa (ms) pausaCorta

al iniciar:
  fijar pausaLarga a 500
  fijar pausaCorta a 250

si logotipo hacia arriba:   // Este bloque detecta el momento en cambia la posición del dispositivo, cuando lo hace se editan las variables "pausaLarga" y "pausaCorta". Al inclinar el dispositivo hacia esta dirección, la velocidad de latido aumentará ya que los tiempos son más cortos.
  fijar pausaLarga a 100
  fijar pausaCorta a 10

si logotipo hacia abajo:   // Este bloque detecta el momento en cambia la posición del dispositivo, cuando lo hace se editan las variables "pausaLarga" y "pausaCorta". Al inclinar el dispositivo hacia esta dirección, la velocidad de latido disminuirá ya que los tiempos son más largos.
  fijar pausaLarga a 500
  fijar pausaCorta a 250
