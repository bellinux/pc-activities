ACTIVIDAD: 17.2-cinematic-power-on
CODE: luz-encendida-ruido-animacion
COMPONENTES: NivelRuido1=NoiseLevel, DibujoLED2=MatrixDraw

protobject_NivelRuido1_change  property=suddenIncrease
  ESTABLECER var:estado = no(var:estado)
  SI
    [si var:estado]
    CONTAR CON filas DE 1 A 7 PASO 1
        ; // Este 'contar con' da a 'filas' los valores del 1 al 7 y ejecuta lo de dentro por cada valor, recorriendo las filas una a una.
      CONTAR CON columnas DE 1 A 7 PASO 1
          ; // Este 'contar con' da a 'columnas' los valores del 1 al 7 dentro de cada fila, así que dibuja 7x7 = 49 veces: una por cada led.
        DibujoLED2_drawOnXY_set_2  VAR=var:columnas  VAR2=var:filas  VAR3=colour_picker(COLOUR=#ff4444)
        ESPERAR  ms  VAR=10
    [sino]
    CONTAR CON filas DE 1 A 7 PASO 1
        ; // Este 'contar con' recorre las filas del 1 al 7 para apagar la luz de cada una.
      CONTAR CON columnas DE 1 A 7 PASO 1
          ; // Dentro de cada fila, este 'contar con' da a 'columnas' los valores del 1 al 7 y apaga el led de cada columna.
        DibujoLED2_drawOnXY_set_2  VAR=var:columnas  VAR2=var:filas  VAR3=colour_picker(COLOUR=#000000)
        ESPERAR  ms  VAR=10

AL INICIAR
  ESTABLECER var:estado = FALSE
