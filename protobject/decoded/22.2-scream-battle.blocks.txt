repetir para siempre:
  si partidaEnCurso:
    establecer columna a 2
    establecer fila a 2
    repetir mientras (ruido in NivelRuido2 ≥ 5):   // 'repetir mientras' da vueltas mientras el ruido sea 5 o más, o sea mientras dura el grito, y sale cuando te callas.
      dibuja en x columna y fila con color #ff4444 en DibujoLED3   // Encendemos un LED rojo en la columna y la fila de ahora: cuanto más largo es el grito, más LEDs se encienden, hasta llenar la matriz.
      retraso de 200 milisegundos   // 'retraso de 200 milisegundos' espera un momento antes del siguiente LED: cada LED encendido vale 200 milisegundos de grito, así que la pantalla se llena en 5 segundos.
      establecer columna a (columna + 1)
      si (columna > 6):
        establecer columna a 2
        establecer fila a (fila + 1)
      si (fila > 6):
        establecer fila a 2
    tocar nota do4 en TecladoMusical1   // Cuando termina el grito suenan tres notas que suben (do, mi, sol): la última se queda 5 segundos mientras miras el resultado.
    retraso de 200 milisegundos
    tocar nota mi4 en TecladoMusical1
    retraso de 200 milisegundos
    tocar nota sol4 en TecladoMusical1
    retraso de 5000 milisegundos
    detener en TecladoMusical1
    borrar dibujo en DibujoLED3   // Apagamos toda la matriz para dejarla lista para el próximo grito.
    establecer partidaEnCurso a falso

al iniciar el programa:
  establecer partidaEnCurso a falso   // Creamos la variable 'partidaEnCurso' y la ponemos en falso: al empezar, no hay ninguna partida en curso.

  establecer partidaEnCurso a verdadero
