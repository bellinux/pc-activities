ACTIVIDAD: 19.1-digital-hot-potato
CODE: papas-caliente-juego
COMPONENTES: TecladoMusical3=PlayKeyboard, BotónTáctil3=TouchButton

REPETIR PARA SIEMPRE (bucle principal)
  SI
    [si var:listo]
    ESTABLECER var:tiempoRestante = aleatorio(5..15)
        ; // Elige al azar cuántos segundos dura la ronda, entre 5 y 15: nadie sabe cuándo va a explotar la papa.
    MIENTRAS (WHILE) (var:tiempoRestante > 0)
        ; // Mientras quede tiempo, en cada vuelta suena un 'tic', se resta 1 a 'tiempoRestante' y se espera un segundo.
      TecladoMusical3_play_set_1  VAR=control_note(VAR=56)
      AÑADIR -1 A var:tiempoRestante
      ESPERAR  ms  VAR=200
      TecladoMusical3_stop_set_3
      ESPERAR  ms  VAR=800
    TecladoMusical3_play_set_1  VAR=control_note(VAR=63)
    ESPERAR  ms  VAR=500
    TecladoMusical3_play_set_1  VAR=control_note(VAR=62)
    ESPERAR  ms  VAR=500
    TecladoMusical3_play_set_1  VAR=control_note(VAR=61)
    ESPERAR  ms  VAR=500
    TecladoMusical3_play_set_1  VAR=control_note(VAR=60)
    ESPERAR  ms  VAR=1000
    TecladoMusical3_stop_set_3
    ESTABLECER var:listo = FALSE
        ; // La ronda terminó: 'listo' vuelve a falso. Si alguien tocó la pantalla durante la ronda, ese toque no cuenta y el juego espera un toque nuevo.

protobject_BotónTáctil3_change  property=pressed
    ; // Al tocar la pantalla, 'listo' pasa a verdadero: así el bucle principal sabe que tiene que empezar una ronda.
  ESTABLECER var:listo = TRUE
